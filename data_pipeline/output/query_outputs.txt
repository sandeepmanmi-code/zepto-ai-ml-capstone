Q6 - JOIN
================================================================================
SELECT
            c.category_name,
            b.title,
            b.price_gbp,
            b.price_inr,
            b.rating,
            b.in_stock
        FROM books AS b
        JOIN categories AS c
            ON b.category_id = c.category_id
        ORDER BY c.category_name, b.rating DESC, b.title
        LIMIT 20;
